2. Custom Screen Implementation (NotebookScreen.java)
Java
package com.example.notebook.client.gui;

import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.client.gui.components.Button;
import net.minecraft.client.gui.components.MultiLineEditBox;
import net.minecraft.client.gui.screens.Screen;
import net.minecraft.network.chat.Component;
import net.minecraft.resources.ResourceLocation;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;

public class NotebookScreen extends Screen {
    private static final ResourceLocation BOOK_TEXTURE = new ResourceLocation("minecraft", "textures/gui/book.png");
    private static final int IMAGE_WIDTH = 192;
    private static final int IMAGE_HEIGHT = 192;

    private final Path notesDir;
    private int currentVolume = 1;
    private int currentPage = 1;
    private final int maxPages = 10;

    private MultiLineEditBox textBox;

    public NotebookScreen(int volume) {
        super(Component.literal("Personal Notebook"));
        this.currentVolume = volume;
        this.notesDir = Minecraft.getInstance().gameDirectory.toPath().resolve("config/notebook/notes");
        ensureDirectoryExists();
    }

    @Override
    protected void init() {
        int leftPos = (this.width - IMAGE_WIDTH) / 2;
        int topPos = (this.height - IMAGE_HEIGHT) / 2;

        // Editable Multi-Line Text Component
        this.textBox = new MultiLineEditBox(
            this.font, 
            leftPos + 20, 
            topPos + 20, 
            140, 
            130, 
            Component.literal("Page Content"), 
            Component.empty()
        );
        
        // Load stored text from /config
        this.textBox.setValue(loadPageText(currentVolume, currentPage));
        this.addRenderableWidget(this.textBox);

        // Previous Page Button
        this.addRenderableWidget(Button.builder(Component.literal("<"), btn -> changePage(-1))
            .bounds(leftPos + 20, topPos + 155, 20, 20)
            .build());

        // Next Page Button
        this.addRenderableWidget(Button.builder(Component.literal(">"), btn -> changePage(1))
            .bounds(leftPos + 150, topPos + 155, 20, 20)
            .build());
    }

    private void changePage(int delta) {
        saveCurrentPage();
        int target = currentPage + delta;
        if (target >= 1 && target <= maxPages) {
            currentPage = target;
            this.textBox.setValue(loadPageText(currentVolume, currentPage));
        }
    }

    private void saveCurrentPage() {
        if (this.textBox == null) return;
        Path pageFile = getPagePath(currentVolume, currentPage);
        try {
            Files.writeString(pageFile, this.textBox.getValue());
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private String loadPageText(int volume, int page) {
        Path pageFile = getPagePath(volume, page);
        if (Files.exists(pageFile)) {
            try {
                return Files.readString(pageFile);
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        return "";
    }

    private Path getPagePath(int volume, int page) {
        return notesDir.resolve(String.format("volume_%d_page_%d.txt", volume, page));
    }

    private void ensureDirectoryExists() {
        try {
            Files.createDirectories(notesDir);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    @Override
    public void render(GuiGraphics guiGraphics, int mouseX, int mouseY, float partialTick) {
        this.renderBackground(guiGraphics);
        
        // Render Vanilla Book Texture
        int leftPos = (this.width - IMAGE_WIDTH) / 2;
        int topPos = (this.height - IMAGE_HEIGHT) / 2;
        guiGraphics.blit(BOOK_TEXTURE, leftPos, topPos, 0, 0, IMAGE_WIDTH, IMAGE_HEIGHT);

        // Render Current Page Indicator
        String pageStr = currentPage + " / " + maxPages;
        int strWidth = this.font.width(pageStr);
        guiGraphics.drawString(this.font, pageStr, leftPos + (IMAGE_WIDTH / 2) - (strWidth / 2), topPos + 160, 0x404040, false);

        super.render(guiGraphics, mouseX, mouseY, partialTick);
    }

    @Override
    public void onClose() {
        saveCurrentPage(); // Auto-save when closing ESC
        super.onClose();
    }

    @Override
    public boolean isPauseScreen() {
        return false; // Game continues running in background
    }
}

Instead of splitting files on disk, you store one master file per volume (e.g., config/notebook/notes/volume_1.txt or .json) and let your Java Screen dynamically chunk the text into virtual pages at runtime.

Page Pagination & Formatting Rules
You can establish a parsing pipeline that breaks raw text into pages using a simple character counter + layout cost system:

Base Character Capacity: ~640 characters fit cleanly on a standard 1-page book layout using standard font height (9px).

Manual Page Breaks: Use a simple tag like [page] or --- in the text. When the parser hits this, it immediately starts a new page regardless of character count.

Inline Images: Use custom tags like [img:image_name.png|half] or [img:image_name.png|full].

A half image reduces the page's character allowance by 50% (~320 chars).

A full image consumes the entire page (0 chars available for text).

Hard Page Cap: Limit the parsed list size to 64 pages (if (pages.size() >= 64) break;).

Java Logic: Text-to-Page Parser
Instead of forcing players to manually split files, write a parser method in your GUI screen that reads the master string and splits it into a List<NotebookPage>:

Java
import java.util.ArrayList;
import java.util.List;

public record NotebookPage(String text, String imagePath, ImageSize imageSize) {}
public enum ImageSize { NONE, HALF, FULL }

public class PageParser {
    private static final int MAX_CHARS_PER_PAGE = 640;
    private static final int MAX_PAGES = 64;

    public static List<NotebookPage> parseVolume(String rawContent) {
        List<NotebookPage> pages = new ArrayList<>();
        // Split manual page breaks first
        String[] manualSections = rawContent.split("\\[page\\]|---");

        for (String section : manualSections) {
            if (pages.size() >= MAX_PAGES) break;
            
            // Extract image tag if present (e.g. [img:flower.png|half])
            ImageInfo img = extractImageTag(section);
            String cleanText = section.replaceAll("\\[img:[^\\]]+\\]", "").trim();

            int capacity = MAX_CHARS_PER_PAGE;
            if (img.size == ImageSize.HALF) capacity /= 2;
            else if (img.size == ImageSize.FULL) capacity = 0;

            // Split remaining text into page chunks based on calculated capacity
            if (capacity > 0 && cleanText.length() > capacity) {
                int index = 0;
                while (index < cleanText.length() && pages.size() < MAX_PAGES) {
                    int end = Math.min(index + capacity, cleanText.length());
                    pages.add(new NotebookPage(cleanText.substring(index, end), img.path, img.size));
                    index = end;
                    // Subsequent auto-overflow pages don't carry the image
                    img = new ImageInfo(null, ImageSize.NONE); 
                    capacity = MAX_CHARS_PER_PAGE;
                }
            } else {
                pages.add(new NotebookPage(cleanText, img.path, img.size));
            }
        }
        return pages;
    }

    private record ImageInfo(String path, ImageSize size) {}

    private static ImageInfo extractImageTag(String text) {
        // Regex / string search for [img:filename.png|half]
        if (text.contains("[img:")) {
            int start = text.indexOf("[img:") + 5;
            int end = text.indexOf("]", start);
            if (end > start) {
                String[] parts = text.substring(start, end).split("\\|");
                ImageSize size = parts.length > 1 && parts[1].equalsIgnoreCase("full") 
                    ? ImageSize.FULL : ImageSize.HALF;
                return new ImageInfo(parts[0], size);
            }
        }
        return new ImageInfo(null, ImageSize.NONE);
    }
}

To load arbitrary PNG files from outside your mod’s assets (like .minecraft/config/notebook/images/), you convert the raw image file into a NativeImage, register it as a DynamicTexture with Minecraft's TextureManager, and render the resulting ResourceLocation.

Image Loader Helper Class (LocalImageLoader.java)
This utility handles reading PNG bytes, generating a DynamicTexture, and caching the registered ResourceLocation so you don't reload the image from disk on every render frame.

Java
package com.example.notebook.client.util;

import com.mojang.blaze3d.platform.NativeImage;
import net.minecraft.client.Minecraft;
import net.minecraft.client.renderer.texture.DynamicTexture;
import net.minecraft.resources.ResourceLocation;

import java.io.FileInputStream;
import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.Map;

public class LocalImageLoader {
    private static final Map<String, ResourceLocation> TEXTURE_CACHE = new HashMap<>();

    /**
     * Loads a PNG from .minecraft/config/notebook/images/filename
     * Returns a registered ResourceLocation ready for GuiGraphics.blit()
     */
    public static ResourceLocation getOrLoadTexture(String fileName) {
        if (TEXTURE_CACHE.containsKey(fileName)) {
            return TEXTURE_CACHE.get(fileName);
        }

        Path imagePath = Minecraft.getInstance().gameDirectory.toPath()
                .resolve("config/notebook/images").resolve(fileName);

        if (!Files.exists(imagePath)) {
            return null; // File missing
        }

        try (InputStream stream = new FileInputStream(imagePath.toFile())) {
            // 1. Read PNG file into Minecraft's NativeImage format
            NativeImage nativeImage = NativeImage.read(stream);

            // 2. Wrap NativeImage into a DynamicTexture
            DynamicTexture dynamicTexture = new DynamicTexture(nativeImage);

            // 3. Register the texture with Minecraft's TextureManager
            String cleanName = fileName.toLowerCase().replaceAll("[^a-z0-9_.-]", "_");
            ResourceLocation location = new ResourceLocation("notebook", "dynamic/image_" + cleanName);
            
            Minecraft.getInstance().getTextureManager().register(location, dynamicTexture);

            // 4. Cache and return
            TEXTURE_CACHE.put(fileName, location);
            return location;

        } catch (Exception e) {
            e.printStackTrace();
            return null;
        }
    }

    /**
     * Call when reloading configs or closing the GUI to avoid memory leaks.
     */
    public static void clearCache() {
        TextureManager manager = Minecraft.getInstance().getTextureManager();
        for (ResourceLocation location : TEXTURE_CACHE.values()) {
            manager.release(location);
        }
        TEXTURE_CACHE.clear();
    }
}

@Override
public void render(GuiGraphics guiGraphics, int mouseX, int mouseY, float partialTick) {
    this.renderBackground(guiGraphics);

    int leftPos = (this.width - IMAGE_WIDTH) / 2;
    int topPos = (this.height - IMAGE_HEIGHT) / 2;

    // 1. Render Book GUI Background
    guiGraphics.blit(BOOK_TEXTURE, leftPos, topPos, 0, 0, IMAGE_WIDTH, IMAGE_HEIGHT);

    // Get active page data
    NotebookPage currentPageData = pages.get(currentPage - 1);

    // 2. Render Dynamic PNG if present
    if (currentPageData.imagePath() != null) {
        ResourceLocation customImg = LocalImageLoader.getOrLoadTexture(currentPageData.imagePath());

        if (customImg != null) {
            if (currentPageData.imageSize() == ImageSize.FULL) {
                // Draws a 120x120 pixel image taking up the entire inner page space
                guiGraphics.blit(customImg, leftPos + 36, topPos + 20, 0, 0, 120, 120, 120, 120);
            } else if (currentPageData.imageSize() == ImageSize.HALF) {
                // Draws a 120x60 image occupying the top half of the page
                guiGraphics.blit(customImg, leftPos + 36, topPos + 20, 0, 0, 120, 60, 120, 60);
            }
        }
    }

    super.render(guiGraphics, mouseX, mouseY, partialTick);
}
blit Parameter Breakdown
Java
guiGraphics.blit(texture, x, y, u, v, width, height, textureWidth, textureHeight);
x, y: Screen coordinates where the top-left corner of the image will be drawn.

u, v: Texture offset coordinates (set to 0, 0 for raw PNGs).

width, height: The rendered size on the screen (in pixels).

textureWidth, textureHeight: Scale factors matching the original file resolution so Minecraft scales the PNG dynamically without cropping it.
